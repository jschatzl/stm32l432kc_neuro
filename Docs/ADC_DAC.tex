\begin{tikzpicture}[
    >=Stealth,
    line width=0.6pt,
    node distance=0.55cm,
    every node/.style={font=\small},
    terminator/.style={
        ellipse,
        draw,
        minimum width=2.6cm,
        minimum height=0.85cm,
        align=center
    },
    process/.style={
        rounded rectangle,
        draw,
        minimum width=4.1cm,
        minimum height=0.9cm,
        align=center
    },
    decision/.style={
        diamond,
        draw,
        aspect=2.0,
        minimum width=2.8cm,
        minimum height=1.0cm,
        align=center,
        inner sep=1pt
    },
    arrow/.style={->}
]

% Main flow
\node[terminator] (start) {Start};
\node[process, below=of start] (sum) {Calculate weighted\\input sum};
\node[process, below=of sum] (savez) {Save weighted input\\sum};
\node[process, below=of savez] (activation) {Calculate activation};
\node[process, below=of activation] (savea) {Save activation};
\node[decision, below=0.75cm of savea] (layerdone) {Layer done?};
\node[decision, below=0.9cm of layerdone] (lastlayer) {Last layer?};
\node[terminator, below=0.75cm of lastlayer] (end) {End};
\node[terminator, right=2.48cm of lastlayer] (next) {Next layer};
\node[terminator, above right={3.28cm and 2.11cm} of lastlayer] (nextN) {Next neuron};

% Main arrows
\draw[arrow] (start) -- (sum);
\draw[arrow] (sum) -- (savez);
\draw[arrow] (savez) -- (activation);
\draw[arrow] (activation) -- (savea);
\draw[arrow] (savea) -- (layerdone);
\draw[arrow] (layerdone) -- node[left] {true} (lastlayer);
\draw[arrow] (lastlayer) -- node[left] {true} (end);

% Layer loop: false returns to weighted sum

% Next-layer loop
\draw[arrow] (lastlayer.east) -- node[above] {false} (next.west);
\draw[arrow] (next.north) -- ++(0,7.94) |- (1.94,-1.45);

  \draw[arrows=Stealth-] (nextN.south) -- node[above=0pt, pos=0.71]{false} |- (layerdone.east);
  \draw[->] (nextN.north) -- |- (1.94,-1.45);
\end{tikzpicture}
